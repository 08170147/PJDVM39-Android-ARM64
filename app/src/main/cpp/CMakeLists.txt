cmake_minimum_required(VERSION 3.22.1)
project(pjdvm39_port LANGUAGES C CXX)
add_library(pjdvm39_port SHARED native-lib.cpp)
target_compile_features(pjdvm39_port PRIVATE cxx_std_20)
target_compile_options(pjdvm39_port PRIVATE -Wall -Wextra -Wpedantic)
find_library(log-lib log)
find_library(android-lib android)
find_library(VULKAN_LIB vulkan REQUIRED)
target_link_libraries(pjdvm39_port PRIVATE ${log-lib} ${android-lib} ${VULKAN_LIB})
target_compile_definitions(pjdvm39_port PRIVATE PJDV_HAS_VULKAN_LOADER=1)
